``` ### 2. ไฟล์ระบบหลังบ้าน (Google Apps Script) นำโค้ดชุดนี้ไปวางไว้ในเมนู **Extensions > Apps Script** ของไฟล์ Google Sheet ของคุณ ```javascript:ระบบหลังบ้าน:Code.gs // ฟังก์ชันสำหรับรับ Request จากหน้าเว็บ function doPost(e) { try { const data = JSON.parse(e.postData.contents); const action = data.action; let result = {}; setupSheets(); // ตรวจสอบการสร้างชีตอัตโนมัติ if (action === "login") { result = loginUser(data.username, data.password); } else if (action === "getSeats") { result = getSeats(data.center); } else if (action === "bookSeats") { result = bookSeats(data.center, data.seats, data.userDetails); } else if (action === "getAllBookings") { result = getAllBookings(); } else if (action === "deleteBooking") { result = deleteBooking(data.rowId); } else { result = { status: "error", message: "Unknown action" }; } return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } } // สร้างชีต Bookings ถ้ายังไม่มี (แยกออกมาจากชีต User อัตโนมัติ) function setupSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss.getSheetByName("Bookings")) { const sheet = ss.insertSheet("Bookings"); sheet.appendRow(["RowID", "Timestamp", "Center", "SeatID", "Title", "FirstName", "LastName", "Phone", "RefCode"]); sheet.getRange("A1:I1").setFontWeight("bold").setBackground("#d9ead3"); } } // ตรวจสอบข้อมูลล็อกอินจากชีต (อ้างอิงโครงสร้าง: A=Name, B=LastName, C=User, D=Password) function loginUser(username, password) { // รหัสไฟล์ Google Sheet ตามที่คุณแนบลิงก์มา const sheetId = '1A2EHKMSeLs-lMgc86R0koydo4dfBrFu1w7i0-1GKBKI'; const ss = SpreadsheetApp.openById(sheetId); // สมมติว่าข้อมูลผู้ใช้อยู่ในชีตแรกสุด const sheet = ss.getSheets()[0]; const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { const fname = data[i][0]; // Name const lname = data[i][1]; // LastName const user = data[i][2]; // User const pass = data[i][3]; // Password // แปลงให้เป็น string ก่อนเช็ค เพื่อป้องกันปัญหารหัสผ่านตัวเลข if (user.toString() === username && pass.toString() === password) { return { status: "success", data: { role: user.toString(), // ใช้ Username เป็น Role เบื้องต้น fname: fname, lname: lname } }; } } return { status: "error", message: "ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง" }; } function getSeats(center) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Bookings"); const data = sheet.getDataRange().getValues(); let seatsData = {}; for (let i = 1; i < data.length; i++) { if (data[i][2] === center) { seatsData[data[i][3]] = { status: 'booked' }; } } return { status: "success", data: seatsData }; } function bookSeats(center, seats, userDetails) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Bookings"); const timestamp = new Date(); const baseRef = "BK-" + timestamp.getTime().toString().slice(-5); seats.forEach((seatId, index) => { let refCode = `${baseRef}-${index+1}`; let rowId = Utilities.getUuid(); // สร้าง ID เฉพาะสำหรับแถวนั้นๆ เพื่อให้ลบง่าย sheet.appendRow([ rowId, timestamp, center, seatId, userDetails.title, userDetails.fname, userDetails.lname, userDetails.phone, refCode ]); }); return { status: "success", baseRefCode: baseRef }; } function getAllBookings() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Bookings"); const data = sheet.getDataRange().getValues(); let bookings = []; for (let i = 1; i < data.length; i++) { bookings.push({ rowId: data[i][0], timestamp: data[i][1], center: data[i][2], seatId: data[i][3], title: data[i][4], fname: data[i][5], lname: data[i][6], phone: data[i][7], refCode: data[i][8] }); } return { status: "success", data: bookings }; } function deleteBooking(rowId) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Bookings"); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === rowId) { sheet.deleteRow(i + 1); return { status: "success" }; } } return { status: "error", message: "ไม่พบข้อมูล" }; }
